智能客服数据模型 ER DIAGRAM
erDiagram
    USER ||--o{ SESSION : "发起"
    SESSION ||--o{ MESSAGE : "包含"
    SESSION ||--o| TICKET : "转人工"
    AGENT ||--o{ TICKET : "处理"
    KNOWLEDGE ||--o{ MESSAGE : "召回"

    USER {
        int id PK "用户ID"
        string nickname "昵称"
        string phone "手机号"
        string channel "来源渠道"
        datetime created_at "注册时间"
    }

    SESSION {
        int id PK "会话ID"
        int user_id FK "用户ID"
        string channel "接入渠道"
        string status "会话状态"
        datetime started_at "开始时间"
        datetime ended_at "结束时间"
    }

    MESSAGE {
        int id PK "消息ID"
        int session_id FK "会话ID"
        string role "user / assistant / agent"
        text content "消息内容"
        string intent "识别意图"
        datetime created_at "发送时间"
    }

    KNOWLEDGE {
        int id PK "知识ID"
        string title "标题"
        string category "分类"
        text content "知识内容"
        int hit_count "召回次数"
        datetime updated_at "更新时间"
    }

    TICKET {
        int id PK "工单ID"
        int session_id FK "会话ID"
        int agent_id FK "坐席ID"
        string status "处理状态"
        string priority "优先级"
        datetime created_at "创建时间"
    }

    AGENT {
        int id PK "坐席ID"
        string name "姓名"
        string team "所属团队"
        boolean online "在线状态"
    }
流程图覆盖接入、理解、检索、生成、转人工与知识沉淀全链路。 ER 图包含用户、会话、消息、知识、工单、坐席六类核心实体。
